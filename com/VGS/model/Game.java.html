1    package com.VGS.model;
2    
3    public class Game {
4        private long id;
5        private String title;
6        private String genre;
7        private String platform;
8        private boolean completed;
9    
10       /* Constructor: Game 
11        * 
12        * Creates a new Game object with the given attributes. 
13        * Validates input to ensure ID is positive, title and platform 
14        * are not blank, and genre contains only letters. 
15        * 
16        *  - long id: Unique identifier for the game (must be positive) 
17        *  - String title: Name of the game (cannot be null or blank) 
18        *  - String genre: Game genre (letters only) 
19        *  - String platform: Platform for the game (cannot be null or blank) 
20        *  - boolean completed: Whether the game has been completed 
21        */
22       public Game(long id, String title, String genre, String platform, boolean completed) {
23           if (id <= 0) throw new IllegalArgumentException("ID must be a positive number.");
24           if (title == null || title.isBlank()) throw new IllegalArgumentException("Title required.");
25           if (genre == null || genre.isBlank() || !genre.matches("[a-zA-Z ]+"))
26               throw new IllegalArgumentException("Genre must only contain letters.");
27           if (platform == null || platform.isBlank()) throw new IllegalArgumentException("Platform required.");
28   
29           this.id = id;
30           this.title = title.trim();
31           this.genre = genre.trim();
32           this.platform = platform.trim();
33           this.completed = completed;
34       }
35   
36       // Getters
37       public long getId() { return id; }
38       public String getTitle() { return title; }
39       public String getGenre() { return genre; }
40       public String getPlatform() { return platform; }
41       public boolean isCompleted() { return completed; }
42   
43       /* Method: setTitle 
44        * 
45        * Updates the title of the game if the new title is valid. 
46        *  - String title: New title for the game (cannot be null or blank) 
47        */
48       public void setTitle(String title) {
49           if (title != null && !title.isBlank()) this.title = title.trim();
50       }
51   
52       /* Method: setGenre 
53        * 
54        * Updates the genre of the game if the new genre is valid. 
55        * Prints an error message if the genre contains invalid characters. 
56        *  - String genre: New genre (letters only) 
57        */
58       public void setGenre(String genre) {
59           if (genre != null && !genre.isBlank() && genre.matches("[a-zA-Z ]+"))
60               this.genre = genre.trim();
61           else
62               System.out.println("Invalid genre. Only letters allowed.");
63       }
64   
65       /* Method: setPlatform 
66        * 
67        * Updates the platform of the game if the new value is valid. 
68        *  - String platform: New platform (cannot be null or blank) 
69        */
70       public void setPlatform(String platform) {
71           if (platform != null && !platform.isBlank()) this.platform = platform.trim();
72       }
73   
74       /* Method: setCompleted 
75        * 
76        * Updates the completion status of the game. 
77        *  - boolean completed: True if the game is completed, false otherwise 
78        */
79       public void setCompleted(boolean completed) {
80           this.completed = completed;
81       }
82   
83       /* Method: toString 
84        * 
85        * Returns a formatted string containing all the game information. 
86        * 
87        * Arguments: none 
88        * Return: String - formatted game details 
89        */
90       @Override
91       public String toString() {
92           return "ID: " + id +
93                   ", Title: " + (title != null ? title : "N/A") +
94                   ", Genre: " + (genre != null ? genre : "N/A") +
95                   ", Platform: " + (platform != null ? platform : "N/A") +
96                   ", Completed: " + completed;
97       }
98   }
