1    package com.VGS.service;
2    
3    import com.VGS.App;
4    import com.VGS.model.Game;
5    import com.VGS.repository.Gamerepository;
6    
7    import java.io.File;
8    import java.util.List;
9    import java.util.Scanner;
10   
11   /* Provides the business logic for the Video Game Collection System. 
12    * Acts as a service layer between the user interface (App) and the repository (Gamerepository). 
13    * Handles adding, removing, updating, viewing, and marking games as completed. 
14    * Also loads games safely from a text file. 
15    */
16   
17   import org.springframework.beans.factory.annotation.Value;
18   import org.springframework.stereotype.Service;
19   
20   @Service
21   public class GameService {
22   
23       private final Gamerepository repository;
24       private final String dbPath;
25   
26       // Added dbPath in order for the database to connect smoothly through the GUI
27       public GameService(Gamerepository repository, String dbPath) {
28           this.repository = repository;
29           this.dbPath = dbPath;
30       }
31   
32       // Add a game to SQLite
33       public boolean addGame(Game game) {
34           if (game == null || existsById(game.getId())) {
35               return false; // prevents null or duplicate ID
36           }
37           try {
38               return repository.addGame(game, dbPath);
39           } catch (Exception e) {
40               System.out.println("Add failed: " + e.getMessage());
41               return false;
42           }
43       }
44   
45       // Remove a game by ID
46       public boolean removeGame(long id) {
47           try {
48               return repository.removeGame(id, dbPath);
49           } catch (Exception e) {
50               System.out.println("Remove failed: " + e.getMessage());
51               return false;
52           }
53       }
54   
55       // View all games
56       public List<Game> viewAllGames() {
57           try {
58               return repository.getAllGames(dbPath);
59           } catch (Exception e) {
60               System.out.println("Load failed: " + e.getMessage());
61               return List.of(); // empty list if error
62           }
63       }
64   
65       // Check if a game exists by ID
66       public boolean existsById(long id) {
67           return findGame(id) != null;
68       }
69   
70       // Update a game by ID
71       public boolean updateGame(long id, String title, String genre, String platform) {
72           Game game = findGame(id);
73           if (game == null) return false; // game not found
74   
75           game.setTitle(title);
76           game.setGenre(genre);
77           game.setPlatform(platform);
78   
79           try {
80               return repository.updateGame(game, dbPath);
81           } catch (Exception e) {
82               System.out.println("Update failed: " + e.getMessage());
83               return false;
84           }
85       }
86   
87       // Mark a game as completed
88       public Game trackCompletion(long id) {
89           // Find the game in the database
90           Game game = repository.findGame(id, dbPath);
91   
92           if (game != null) {
93               // Mark as completed
94               game.setCompleted(true);
95   
96               // Update the database row
97               boolean updated = repository.updateGame(game, dbPath);
98   
99               if (updated) {
100                  return game; // Return updated game
101              } else {
102                  System.out.println("Failed to update game completion in DB for ID: " + id);
103              }
104          }
105          return null; // game not found or DB update failed
106      }
107  
108      // Find a game by ID
109      public Game findGame(long id) {
110          try {
111              return repository.findGame(id, dbPath);
112          } catch (Exception e) {
113              System.out.println("Find failed: " + e.getMessage());
114              return null;
115          }
116      }
117  
118      /* Load games from a text file directly into the SQLite database. 
119       * Each line must be in format: id,title,genre,platform,completed 
120       */
121      public int loadGamesFromFile(String fileName) {
122  
123          // Updated*** to check if DB already has data
124          if (!repository.getAllGames(dbPath).isEmpty()) {
125              System.out.println("Database already contains data. Skipping file load.");
126              return 0;
127          }
128  
129          int count = 0;
130          Scanner fileScanner = null;
131  
132          try {
133              var inputStream = App.class.getClassLoader().getResourceAsStream(fileName);
134              if (inputStream == null) {
135                  System.out.println("File not found in resources: " + fileName);
136                  return 0;
137              }
138  
139              fileScanner = new Scanner(inputStream);
140  
141              while (fileScanner.hasNextLine()) {
142                  String line = fileScanner.nextLine();
143                  String[] parts = line.split(",");
144  
145                  if (parts.length < 5) continue;
146  
147                  try {
148                      long id = Long.parseLong(parts[0].trim());
149                      String title = parts[1].trim();
150                      String genre = parts[2].trim();
151                      String platform = parts[3].trim();
152                      boolean completed = Boolean.parseBoolean(parts[4].trim());
153  
154                      Game game = new Game(id, title, genre, platform, completed);
155  
156                      if (repository.addGame(game, dbPath)) count++;
157  
158                  } catch (Exception e) {
159                      // skip bad line
160                  }
161              }
162  
163          } catch (Exception e) {
164              System.out.println("Error loading file: " + e.getMessage());
165          } finally {
166              if (fileScanner != null) fileScanner.close();
167          }
168  
169          return count;
170      }
171  }
