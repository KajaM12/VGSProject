35                   id INTEGER PRIMARY KEY, 
36                   title TEXT NOT NULL, 
37                   genre TEXT NOT NULL, 
38                   platform TEXT NOT NULL, 
39                   completed BOOLEAN 
40               ); 
41           """);
42   
43               stmt.execute("DELETE FROM games");

45           } catch (Exception e) {
46               System.out.println("Failed to reset DB: " + e.getMessage());

52       /* Test adding a game to the service. 
53        * Verifies that addGame returns true and the game is actually added. 
54        */
55       @Test
56       public void testAddGame() {

58           boolean added = service.addGame(game);
59           assertTrue("Game should be added successfully", added);
60   
61           // Attempt duplicate ID
62           Game duplicate = new Game(1, "Mario", "Platformer", "Switch", false);
63           boolean duplicateAdded = service.addGame(duplicate);
64           assertFalse("Duplicate ID should not be added", duplicateAdded);
65       }

67       /* Test removing a game that exists. 
68        * Ensures removeGame returns true and the game is removed from the list. 
69        */
70       @Test
71       public void testRemoveGame() {
72           service.addGame(new Game(4, "Halo", "FPS", "Xbox", false));
73           boolean removed = service.removeGame(4);
74           assertTrue("Game should be removed", removed);
75   
76           // Attempt to remove non-existing game
77           boolean removedAgain = service.removeGame(999);
78           assertFalse("Removing non-existing game should fail", removedAgain);
