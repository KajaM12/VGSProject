1    package com.VGS.controller;

3    import org.springframework.ui.Model;
4    import org.springframework.web.bind.annotation.ControllerAdvice;
5    import org.springframework.web.bind.annotation.ExceptionHandler;
6    
7    
8        //GlobalExceptionHandler is a centralized error handler for the entire application.
9    @ControllerAdvice
10       public class GlobalExceptionHandler {
11   
12       /* This method that is added handles all exceptions of type Exception (any unhandled errors). 
13        * @param e The exception that occurred. 
14        * @param model Model used to pass data to the view. 
15        * @return The name of the error template to render. 
16        */
17       @ExceptionHandler(Exception.class)
18       public String handleException(Exception e, Model model) {
19           // Add a generic error message to the model for the error page.
20           model.addAttribute("errorMessage", "Something went wrong. Please try again.");
21           return "error"; // Returns the error.html template to display the message
22       }
23   }
